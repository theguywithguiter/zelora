function parsePersonalizationData(value?: string | null) {
  if (!value) return [];
  try {
    const parsed = JSON.parse(value);
    return Array.isArray(parsed)
      ? parsed.filter(
          (item) =>
            item &&
            typeof item.label === "string" &&
            typeof item.value === "string" &&
            item.value
        )
      : [];
  } catch {
    return [];
  }
}

function statusClass(status: string) {
  return `zelora-order-status zelora-order-status-${status.toLowerCase()}`;
}

  async function loadOrders() {
    try {
      setLoading(true);
      setLoadError(null);

      const response = await fetch("/api/admin/orders", {
        method: "GET",
        cache: "no-store",
        headers: {
          Accept: "application/json",
        },
      });

      const raw = await response.text();

      let data: any = null;

      if (raw.trim()) {
        try {
          data = JSON.parse(raw);
        } catch {
          throw new Error(
            `Orders API returned a non-JSON response (HTTP ${response.status}).`
          );
        }
      }

      if (!response.ok) {
        throw new Error(
          data?.message ||
            `Failed to load orders (HTTP ${response.status}).`
        );
      }

      if (!data || data.success !== true) {
        throw new Error(data?.message || "Failed to load orders.");
      }

      setOrders(Array.isArray(data.orders) ? data.orders : []);
    } catch (error) {
      console.error("Failed to load orders:", error);
      setOrders([]);

      setLoadError(
        error instanceof Error
          ? error.message
          : "Failed to load orders."
      );
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    loadOrders();
  }, []);

  async function updateStatus(id: number, status: string) {
    try {
      setUpdating(id);

      const response = await fetch("/api/admin/orders", {
        method: "PATCH",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          id,
          status,
        }),
      });

      const raw = await response.text();

      let data: any = null;

      if (raw.trim()) {
        try {
          data = JSON.parse(raw);
        } catch {
          throw new Error(
            `Order status API returned a non-JSON response (HTTP ${response.status}).`
          );
        }
      }

      if (!response.ok || !data?.success) {
        throw new Error(
          data?.message || `Failed to update order (HTTP ${response.status}).`
        );
      }

      setOrders((current) =>
        current.map((order) =>
          order.id === id
            ? {
                ...order,
                status,
                trackingNumber:
                  data.order?.trackingNumber ?? order.trackingNumber ?? null,
                trackingWebsite:
                  data.order?.trackingWebsite ?? order.trackingWebsite ?? null,
              }
            : order
        )
      );
    } catch (error) {
      console.error("Failed to update order:", error);
      alert("Failed to update order");
    } finally {
      setUpdating(null);
    }
  }

  async function saveTrackingDetails(order: Order) {
    const trackingNumber = String(order.trackingNumber || "").trim();
    const trackingWebsite = String(order.trackingWebsite || "").trim();

    if (!trackingNumber || !trackingWebsite) {
      alert("Please enter both the tracking number and tracking website.");
      return;
    }

    try {
      setSavingTracking(order.id);

      const response = await fetch("/api/admin/orders", {
        method: "PATCH",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          id: order.id,
          status: order.status,
          trackingNumber,
          trackingWebsite,
        }),
      });

      const raw = await response.text();

      let data: any = null;

      if (raw.trim()) {
        try {
          data = JSON.parse(raw);
        } catch {
          throw new Error(
            `Tracking API returned a non-JSON response (HTTP ${response.status}).`
          );
        }
      }

      if (!response.ok || !data?.success) {
        throw new Error(
          data?.message ||
            `Failed to save tracking details (HTTP ${response.status}).`
        );
      }

      setOrders((current) =>
        current.map((currentOrder) =>
          currentOrder.id === order.id
            ? {
                ...currentOrder,
                trackingNumber:
                  data.order?.trackingNumber ?? trackingNumber,
                trackingWebsite:
                  data.order?.trackingWebsite ?? trackingWebsite,
              }
            : currentOrder
        )
      );
    } catch (error) {
      console.error("Failed to save tracking details:", error);
      alert(
        error instanceof Error
          ? error.message
          : "Failed to save tracking details."
      );
    } finally {
      setSavingTracking(null);
    }
  }
