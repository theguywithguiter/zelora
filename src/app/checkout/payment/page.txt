"use client";

import { useEffect, useMemo, useState } from "react";
import { useRouter } from "next/navigation";

const UPI_ID = "9907374537@sbi";
const PAYMENT_WINDOW_SECONDS = 10 * 60;

type PaymentInfo = {
  orderId: string;
  orderNumber: string;
  amount: number;
};

function formatAmount(amount: number) {
  return amount.toLocaleString("en-IN", {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  });
}

function buildUpiUri(info: PaymentInfo) {
  const params = new URLSearchParams({
    pa: UPI_ID,
    am: info.amount.toFixed(2),
    cu: "INR",
    tr: info.orderId,
  });

  return `upi://pay?${params.toString()}`;
}

function buildAppUri(
  app: "phonepe" | "paytm" | "bhim" | "gpay",
  info: PaymentInfo
) {
  const encodedOrderId = encodeURIComponent(info.orderId);
  const amount = info.amount.toFixed(2);
  const pa = encodeURIComponent(UPI_ID);

  switch (app) {
    case "phonepe":
      return `phonepe://pay?pa=${pa}&am=${amount}&cu=INR&tr=${encodedOrderId}`;
    case "paytm":
      return `paytmmp://pay?pa=${pa}&am=${amount}&cu=INR&tr=${encodedOrderId}`;
    case "bhim":
      return `bhim://pay?pa=${pa}&am=${amount}&cu=INR&tr=${encodedOrderId}`;
    case "gpay":
      return `tez://upi/pay?pa=${pa}&am=${amount}&cu=INR&tr=${encodedOrderId}`;
  }
}

export default function UpiPaymentPage() {
  const router = useRouter();
  const [info, setInfo] = useState<PaymentInfo | null>(null);
  const [secondsLeft, setSecondsLeft] = useState(PAYMENT_WINDOW_SECONDS);
  const [loading, setLoading] = useState(true);
  const [completed, setCompleted] = useState(false);

  useEffect(() => {
    try {
      const params = new URLSearchParams(window.location.search);
      const stored = sessionStorage.getItem("zelora-upi-payment");
      const parsedStored = stored ? JSON.parse(stored) : null;

      const orderId = params.get("orderId") || parsedStored?.orderId || "";
      const orderNumber =
        params.get("orderNumber") || parsedStored?.orderNumber || "";
      const amount = Number(
        params.get("amount") || parsedStored?.amount || 0
      );

      if (!orderId || !amount) {
        router.replace("/checkout");
        return;
      }

      const paymentExpiryKey = `zelora-upi-payment-expires-${orderId}`;
      const storedExpiry = Number(
        localStorage.getItem(paymentExpiryKey) || 0
      );

      const expiry =
        storedExpiry > Date.now()
          ? storedExpiry
          : Date.now() + PAYMENT_WINDOW_SECONDS * 1000;

      if (expiry <= Date.now()) {
        localStorage.removeItem(paymentExpiryKey);
        router.replace("/checkout");
        return;
      }

      localStorage.setItem(paymentExpiryKey, String(expiry));

      setInfo({
        orderId,
        orderNumber,
        amount,
      });
      setSecondsLeft(
        Math.max(0, Math.ceil((expiry - Date.now()) / 1000))
      );
      setLoading(false);
    } catch {
      router.replace("/checkout");
    }
  }, [router]);

  useEffect(() => {
    if (loading || completed) return;

    const expiryKey = `zelora-upi-payment-expires-${info?.orderId || ""}`;
    const expiry = Number(localStorage.getItem(expiryKey) || 0);

    const timer = window.setInterval(() => {
      const remaining = Math.max(
        0,
        Math.ceil((expiry - Date.now()) / 1000)
      );

      setSecondsLeft(remaining);

      if (remaining <= 0) {
        window.clearInterval(timer);
        localStorage.removeItem(expiryKey);
        sessionStorage.removeItem("zelora-upi-payment");
        router.replace("/checkout");
      }
    }, 250);

    return () => window.clearInterval(timer);
  }, [loading, completed, info?.orderId, router]);

  const upiUri = useMemo(
    () => (info ? buildUpiUri(info) : ""),
    [info]
  );

  const qrUrl = useMemo(() => {
    if (!upiUri) return "";
    return `https://api.qrserver.com/v1/create-qr-code/?size=420x420&margin=10&data=${encodeURIComponent(
      upiUri
    )}`;
  }, [upiUri]);

  const minutes = Math.floor(secondsLeft / 60)
    .toString()
    .padStart(2, "0");
  const seconds = (secondsLeft % 60).toString().padStart(2, "0");

  function openPaymentApp(app: "phonepe" | "paytm" | "bhim" | "gpay") {
    if (!info) return;

    window.location.href = buildAppUri(app, info);
  }

  function markCompleted() {
    if (!info) return;

    setCompleted(true);
    sessionStorage.setItem(
      "zelora-upi-completed",
      JSON.stringify({
        ...info,
        completedAt: Date.now(),
      })
    );
    sessionStorage.removeItem("zelora-upi-payment");
    localStorage.removeItem(`zelora-upi-payment-expires-${info.orderId}`);
    router.replace(`/checkout/payment/success?orderId=${encodeURIComponent(info.orderId)}`);
  }

  if (loading || !info) {
    return (
      <main style={styles.page}>
        <div style={styles.loading}>Preparing secure UPI payment…</div>
      </main>
    );
  }

  return (
    <main style={styles.page}>
      <div style={styles.glow} />
      <div style={styles.shell}>
        <header style={styles.header}>
          <div>
            <p style={styles.eyebrow}>ZELORA SECURE PAYMENT</p>
            <h1 style={styles.title}>Complete your payment.</h1>
          </div>
          <div style={styles.orderBadge}>
            ORDER ID <strong>{info.orderId}</strong>
          </div>
        </header>

        <section style={styles.card}>
          <div style={styles.amountBlock}>
            <span style={styles.label}>AMOUNT TO PAY</span>
            <strong style={styles.amount}>₹{formatAmount(info.amount)}</strong>
            <span style={styles.note}>
              Scan the QR with any supported UPI app.
            </span>
          </div>

          <div style={styles.divider} />

          <div style={styles.qrWrap}>
            <div style={styles.qrFrame}>
              <img
                src={qrUrl}
                alt="ZELORA UPI payment QR code"
                style={styles.qr}
              />
            </div>
          </div>

          <p style={styles.scanText}>SCAN TO PAY</p>
          <p style={styles.description}>
            Payment note / description: <strong>{info.orderId}</strong>
          </p>

                    <style>{`
            @media (max-width: 600px) {
              .upi-apps {
                flex-wrap: nowrap !important;
                gap: 8px !important;
              }

              .upi-apps > button {
                width: 60px !important;
                height: 60px !important;
                flex: 0 0 60px !important;
              }

              .upi-apps > button img {
                max-width: 46px;
                max-height: 46px;
              }
            }
          `}</style>

<div className="upi-apps" style={styles.apps}>
            <button
              type="button"
              style={{ ...styles.appButton, ...styles.phonePeButton }}
              onMouseEnter={(event) => {
                event.currentTarget.style.transform = "translateY(-3px) scale(1.03)";
                event.currentTarget.style.boxShadow = "0 12px 28px rgba(0,0,0,.28)";
              }}
              onMouseLeave={(event) => {
                event.currentTarget.style.transform = "translateY(0) scale(1)";
                event.currentTarget.style.boxShadow = "0 7px 22px rgba(0,0,0,.18)";
              }}
              onMouseDown={(event) => {
                event.currentTarget.style.transform = "scale(.96)";
              }}
              onMouseUp={(event) => {
                event.currentTarget.style.transform = "translateY(-3px) scale(1.03)";
              }}
              onClick={() => openPaymentApp("phonepe")}
              aria-label="Pay with PhonePe"
            >
              <img
                src="/upi/phonepe.png"
                alt=""
                style={styles.phonePeLogo}
              />
            </button>

            <button
              type="button"
              style={styles.appButton}
              onMouseEnter={(event) => {
                event.currentTarget.style.transform = "translateY(-3px) scale(1.03)";
                event.currentTarget.style.boxShadow = "0 12px 28px rgba(0,0,0,.28)";
              }}
              onMouseLeave={(event) => {
                event.currentTarget.style.transform = "translateY(0) scale(1)";
                event.currentTarget.style.boxShadow = "0 7px 22px rgba(0,0,0,.18)";
              }}
              onMouseDown={(event) => {
                event.currentTarget.style.transform = "scale(.96)";
              }}
              onMouseUp={(event) => {
                event.currentTarget.style.transform = "translateY(-3px) scale(1.03)";
              }}
              onClick={() => openPaymentApp("paytm")}
              aria-label="Pay with Paytm"
            >
              <span style={styles.appLogoFrame}>
                <img
                  src="/upi/paytm.png"
                  alt=""
                  style={styles.appLogo}
                />
              </span>
            </button>

            <button
              type="button"
              style={styles.appButton}
              onMouseEnter={(event) => {
                event.currentTarget.style.transform = "translateY(-3px) scale(1.03)";
                event.currentTarget.style.boxShadow = "0 12px 28px rgba(0,0,0,.28)";
              }}
              onMouseLeave={(event) => {
                event.currentTarget.style.transform = "translateY(0) scale(1)";
                event.currentTarget.style.boxShadow = "0 7px 22px rgba(0,0,0,.18)";
              }}
              onMouseDown={(event) => {
                event.currentTarget.style.transform = "scale(.96)";
              }}
              onMouseUp={(event) => {
                event.currentTarget.style.transform = "translateY(-3px) scale(1.03)";
              }}
              onClick={() => openPaymentApp("bhim")}
              aria-label="Pay with BHIM UPI"
            >
              <span style={styles.appLogoFrame}>
                <img
                  src="/upi/bhim.png"
                  alt=""
                  style={styles.appLogo}
                />
              </span>
            </button>

            <button
              type="button"
              style={styles.appButton}
              onMouseEnter={(event) => {
                event.currentTarget.style.transform = "translateY(-3px) scale(1.03)";
                event.currentTarget.style.boxShadow = "0 12px 28px rgba(0,0,0,.28)";
              }}
              onMouseLeave={(event) => {
                event.currentTarget.style.transform = "translateY(0) scale(1)";
                event.currentTarget.style.boxShadow = "0 7px 22px rgba(0,0,0,.18)";
              }}
              onMouseDown={(event) => {
                event.currentTarget.style.transform = "scale(.96)";
              }}
              onMouseUp={(event) => {
                event.currentTarget.style.transform = "translateY(-3px) scale(1.03)";
              }}
              onClick={() => openPaymentApp("gpay")}
              aria-label="Pay with Google Pay"
            >
              <span style={styles.appLogoFrame}>
                <img
                  src="/upi/google-pay.png"
                  alt=""
                  style={styles.appLogo}
                />
              </span>
            </button>
          </div>

          <p style={styles.payByAnyUpi}>PAY BY CLICKING ANY UPI APP</p>

          <div style={styles.timerBlock}>
            <span style={styles.label}>PAYMENT WINDOW</span>
            <strong style={styles.timer}>{minutes}:{seconds}</strong>
            <span style={styles.timerHint}>
              Complete the payment before the timer expires.
            </span>
          </div>

          <button style={styles.completedButton} onClick={markCompleted}>
            COMPLETED <span></span>
          </button>

          <p style={styles.security}>
            ZELORA will not ask for your UPI PIN, OTP or banking password.
          </p>
        </section>
      </div>
    </main>
  );
}
